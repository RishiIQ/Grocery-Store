node_dispatch_hook.js 
// Initialize FreshCart Telemetry Client
const { FreshCartNode } = require('@freshcart/sdk');

const client = new FreshCartNode({
  apiKey: process.env.FRESHCART_API_KEY,
  region: 'eu-west-1'
});

async function dispatchOrder() {
  const telemetryStream = await client.dispatch({
    hubId: 'hub-paris-04',
    temperatureMax: 2.5,
    maxDispatchTimeMins: 15
  });
  console.log(`Telemetry Node Active: ${telemetryStream.id}`);
}

dispatchOrder();
